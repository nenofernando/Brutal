cmake_minimum_required(VERSION 3.22)
project(NFBRInstaller LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 17)
if(NOT NFBR_PAYLOAD_DLL OR NOT NFBR_VERSION)
 message(FATAL_ERROR "pass -DNFBR_PAYLOAD_DLL=<plug-in dll> -DNFBR_VERSION=X.Y.Z")
endif()
configure_file(installer.rc.in ${CMAKE_BINARY_DIR}/installer.rc @ONLY)
add_executable(NFBRInstaller WIN32 installer.cpp ${CMAKE_BINARY_DIR}/installer.rc)
target_compile_definitions(NFBRInstaller PRIVATE NFBR_VERSION="${NFBR_VERSION}" NFBR_WVERSION=L"${NFBR_VERSION}" UNICODE _UNICODE)
target_link_libraries(NFBRInstaller PRIVATE shell32 ole32 advapi32 user32)
target_link_options(NFBRInstaller PRIVATE /MANIFEST:NO)
